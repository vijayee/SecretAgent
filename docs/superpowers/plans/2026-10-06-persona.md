```c
typedef struct persona_record_t {
  unsigned version;
  char* name;
  char* text;
  char* placement;                 /* "first" | "below" */
  struct { char* key; char* text; }* guidance;   /* tool-conditional sections */
  size_t nguidance;
  char** principles;               /* optional; the falsifiability check's */
  size_t nprinciples;
  char** test_spec;                /* the checkable-behavior statements */
  size_t ntest_spec;
} persona_record_t;

int persona_record_load(const char* record_json, persona_record_t** out);
void persona_record_destroy(persona_record_t* r);
/* compose: byte-stable per inputs; the tools = an array of tool-id strings
   (BORROWED); base = the caller's base text (borrowed); returns heap (caller
   frees), NULL on OOM only. The datetime substitution stamps now — the
   record's load warned. */
char* persona_compose(const persona_record_t* record,
                      const char* context_json_or_text,
                      const char* const* tools, size_t ntools,
                      const char* base);
/* the store record's install (STATE, never events: a store batch via the
   caller's root — the frame layer's sync shape is NOT callable here; the
   install rides the CALLER's store actor: keep persona.c PURE — the install
   API lives in frame.c (Task 2) and persona.c owns only the record shapes +
   validation + compose. */
int persona_validate_falsifiable(const persona_record_t* r, char** err_out);
```
